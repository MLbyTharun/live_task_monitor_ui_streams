# Live Task Monitor — detailed runtime requirements
# =================================================
#
# Project name: live-task-monitor
# Python:      >= 3.11 (3.12/3.13 also work; use a virtual environment)
# Platforms:   Linux, macOS, Windows, containers. No GPU is required for the
#              app itself. CPU-only devices are supported. A vision model must
#              be reachable separately (local server or hosted API).
#
# Install inside a virtual environment:
#
#   Linux / macOS:
#     python3 -m venv .venv
#     source .venv/bin/activate
#     python -m pip install --upgrade pip
#     python -m pip install -r requirement.txt
#     python -m pip install -e .
#
#   Windows PowerShell:
#     python -m venv .venv
#     .\.venv\Scripts\Activate.ps1
#     python -m pip install --upgrade pip
#     python -m pip install -r requirement.txt
#     python -m pip install -e .
#
#   Windows Command Prompt:
#     python -m venv .venv
#     .venv\Scripts\activate.bat
#     python -m pip install --upgrade pip
#     python -m pip install -r requirement.txt
#     python -m pip install -e .
#
# `pip` installs transitive dependencies automatically. The project's
# pyproject.toml also declares the same runtime dependencies, so
# `python -m pip install -e .` is sufficient after installing Python; this file
# exists to make the requirement set explicit and reproducible on any device.
#
# Do NOT install `opencv-python` instead of `opencv-python-headless` unless you
# need OpenCV GUI windows. This app only reads frames and does not open windows.
# On systems without a prebuilt OpenCV wheel, see "Any device" notes in
# Readme.md (build tools, FFmpeg, or use Docker).

# ---------------------------------------------------------------------------
# Camera / frame processing
# ---------------------------------------------------------------------------
# `opencv-python-headless` decodes RTSP/HLS/files/webcam, reads/writes JPEG,
# and resizes frames. It bundles FFmpeg on common platforms; no separate
# system FFmpeg install is required for normal use. Keep `<6` to avoid an
# unreviewed next major.
opencv-python-headless>=4.10,<6.0

# OpenCV uses NumPy for frame buffers. `ltm` also uses it for push JPEG
# decoding. Pin below 3 to avoid an unreviewed next major; pip picks a version
# compatible with your Python and OpenCV.
numpy>=1.23,<3.0
